cmake_minimum_required(VERSION 3.16)
project(cpp_experiments LANGUAGES CXX)

# 设置 C++ 标准
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED OFF)
set(CMAKE_CXX_EXTENSIONS OFF)

# 检测编译器 C++ 标准支持能力
# 注意：AppleClang 虽然 __cplusplus 仍是 201703L，但很多 C++23 头文件已可用
macro(check_cxx_standard_support std result_var)
    set(${result_var} TRUE)
    
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        if(${std} EQUAL 23)
            if(CMAKE_CXX_COMPILER_VERSION LESS 13)
                set(${result_var} FALSE)
            endif()
        elseif(${std} EQUAL 20)
            if(CMAKE_CXX_COMPILER_VERSION LESS 10)
                set(${result_var} FALSE)
            endif()
        endif()
    elseif(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        if(${std} EQUAL 23)
            if(CMAKE_CXX_COMPILER_VERSION LESS 16)
                set(${result_var} FALSE)
            endif()
        elseif(${std} EQUAL 20)
            if(CMAKE_CXX_COMPILER_VERSION LESS 13)
                set(${result_var} FALSE)
            endif()
        endif()
    elseif(CMAKE_CXX_COMPILER_ID STREQUAL "AppleClang")
        # AppleClang 15+ 支持 C++23 部分特性（std::expected, std::span 等）
        # 但 __cplusplus 仍报告为 201703L，需要特殊处理
        if(${std} EQUAL 23)
            if(CMAKE_CXX_COMPILER_VERSION LESS 15)
                set(${result_var} FALSE)
            endif()
        elseif(${std} EQUAL 20)
            if(CMAKE_CXX_COMPILER_VERSION LESS 13)
                set(${result_var} FALSE)
            endif()
        endif()
    endif()
endmacro()

# 添加子目录的宏
macro(add_subdirectory_if_supported dir required_standard)
    check_cxx_standard_support(${required_standard} supported)
    if(supported)
        add_subdirectory(${dir})
    else()
        message(STATUS "Skipping ${dir}: compiler does not support C++${required_standard}")
    endif()
endmacro()

# 添加所有实验
add_subdirectory_if_supported(templates 20)
add_subdirectory_if_supported(smart-pointers 20)
add_subdirectory_if_supported(lambda 20)
add_subdirectory_if_supported(move-semantics 20)
add_subdirectory_if_supported(concurrency 20)
add_subdirectory_if_supported(ranges 20)
add_subdirectory_if_supported(coroutines 20)
add_subdirectory_if_supported(format 20)
add_subdirectory_if_supported(std-span 20)
add_subdirectory_if_supported(std-expected 23)
